function analyze(data,width,height,startY=0){
  const columns=new Uint32Array(width);
  let x0=width,y0=height,x1=-1,y1=-1;
  for(let y=startY;y<height;y++)for(let x=0;x<width;x++){
    const k=(y*width+x)*4;
    if(data[k+3]>0&&Math.min(data[k],data[k+1],data[k+2])<220){
      columns[x]++;x0=Math.min(x0,x);y0=Math.min(y0,y);x1=Math.max(x1,x);y1=Math.max(y1,y);
    }
  }
  if(x1<0)throw new Error('No visible character');
  const cuts=[0];
  for(let i=1;i<3;i++){
    const center=width*i/3,radius=Math.round(width/9),gaps=[];
    let start=-1;
    for(let x=Math.floor(center-radius);x<Math.ceil(center+radius);x++){
      if(columns[x]===0){if(start<0)start=x;}
      else if(start>=0){gaps.push([start,x]);start=-1;}
    }
    if(start>=0)gaps.push([start,Math.ceil(center+radius)]);
    const eligible=gaps.filter(([a,b])=>b-a>=8).sort((a,b)=>Math.abs((a[0]+a[1])/2-center)-Math.abs((b[0]+b[1])/2-center));
    if(!eligible.length)throw new Error('No safe gap for view boundary '+i);
    cuts.push(Math.floor((eligible[0][0]+eligible[0][1])/2));
  }
  cuts.push(width);
  return {bounds:[x0,y0,x1+1,y1+1],cuts};
}

}
