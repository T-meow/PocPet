// 按整张图的空白走廊选择切线，避免模型稍偏离等分网格时切到主体。
function findCuts(data,width,height,axis) {
  const length=axis==='x'?width:height,counts=new Uint32Array(length);
  for(let y=0;y<height;y++) for(let x=0;x<width;x++) {
    const k=(y*width+x)*4;
    if(data[k+3]>0 && Math.min(data[k],data[k+1],data[k+2])<220) counts[axis==='x'?x:y]++;
  }
  const cuts=[0];
  for(let i=1;i<4;i++) {
    const expected=length*i/4,radius=Math.floor(length/20),gaps=[];
    let start=-1;
    for(let n=Math.floor(expected-radius);n<=Math.ceil(expected+radius);n++) {
      if(counts[n]===0) {if(start<0)start=n;}
      else if(start>=0) {gaps.push([start,n]);start=-1;}
    }
    if(start>=0)gaps.push([start,Math.ceil(expected+radius)+1]);
    const eligible=gaps.filter(([a,b])=>b-a>=4).sort((a,b)=>Math.abs((a[0]+a[1])/2-expected)-Math.abs((b[0]+b[1])/2-expected));
    if(!eligible.length)throw new Error('No safe grid gutter on '+axis+' boundary '+i);
    cuts.push(Math.floor((eligible[0][0]+eligible[0][1])/2));
  }
  return [...cuts,length];
}

function removeWhiteBackground(data, width, height) {
  const visited = new Uint8Array(width*height), queue = new Int32Array(width*height);
  let tail=0, head=0;
  const visit = i => {
    if (visited[i]) return;
    visited[i]=1;
    const k=i*4, colors=[data[k],data[k+1],data[k+2]];
    if (data[k+3]===0 || (Math.min(...colors)>=238 && Math.max(...colors)-Math.min(...colors)<=24)) queue[tail++]=i;
  };
  for(let x=0;x<width;x++) {visit(x);visit((height-1)*width+x);}
  for(let y=0;y<height;y++) {visit(y*width);visit(y*width+width-1);}
  while(head<tail) {
    const i=queue[head++],x=i%width,y=Math.floor(i/width);
    data.fill(0,i*4,i*4+4);
    if(x>0) visit(i-1); if(x<width-1) visit(i+1);
    if(y>0) visit(i-width); if(y<height-1) visit(i+width);
  }
  let x0=width,y0=height,x1=-1,y1=-1,foreground=0,edgePixels=0;
  for(let y=0;y<height;y++) for(let x=0;x<width;x++) if(data[(y*width+x)*4+3]>0) {
    foreground++;x0=Math.min(x0,x);y0=Math.min(y0,y);x1=Math.max(x1,x);y1=Math.max(y1,y);
    if(x===0||y===0||x===width-1||y===height-1) edgePixels++;
  }
  if(!foreground) throw new Error('Empty icon');
  return {foregroundPixels:foreground,removedBackgroundPixels:tail,bounds:[x0,y0,x1+1,y1+1],minMargin:Math.min(x0,y0,width-x1-1,height-y1-1),edgePixels};
}
